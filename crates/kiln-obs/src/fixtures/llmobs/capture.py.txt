"""What dd-trace-py 4.16 sends of an LLM app (`docs/observe.md` §8.12), captured on the container's
loopback: a fake model server (OpenAI's and Anthropic's APIs, fixed answers) on 127.0.0.1:18080,
an Agent stand-in on 127.0.0.1:8126 that keeps every request (its path, headers and body) under
/out/<mode>/. Modes: apm_agent (APM on, the Agent's /info listing /evp_proxy/v2: events ride the
APM spans' meta_struct), proxy (APM off: events through the Agent's EVP proxy), agentless (the
intake's origin overridden to the stand-in)."""
import http.server, json, os, socketserver, sys, threading, time

MODE = sys.argv[1]
OUT = f"/out/{MODE}"
os.makedirs(OUT, exist_ok=True)
seq = [0]
lock = threading.Lock()


class Quiet(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, *a):
        pass

    def body(self):
        n = int(self.headers.get("Content-Length") or 0)
        return self.rfile.read(n) if n else b""

    def answer(self, code, obj):
        b = json.dumps(obj).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(b)))
        self.end_headers()
        self.wfile.write(b)


class Agent(Quiet):
    def keep(self, body):
        with lock:
            seq[0] += 1
            n = seq[0]
        name = f"{n:03d}-{self.command}-{self.path.strip('/').replace('/', '_')[:80]}"
        with open(f"{OUT}/{name}.body", "wb") as f:
            f.write(body)
        with open(f"{OUT}/{name}.json", "w") as f:
            json.dump({"method": self.command, "path": self.path, "headers": dict(self.headers.items())}, f, indent=1)

    def do_GET(self):
        self.keep(b"")
        if self.path == "/info":
            return self.answer(200, {"version": "7.84.0", "endpoints": ["/v0.4/traces", "/v0.5/traces", "/v0.6/stats", "/evp_proxy/v2/", "/profiling/v1/input", "/telemetry/proxy/"],
                                     "client_drop_p0s": False, "span_meta_structs": True, "config": {"default_env": "none"}})
        self.answer(404, {})

    def do_PUT(self):
        self.do_POST()

    def do_POST(self):
        self.keep(self.body())
        if "/traces" in self.path:
            return self.answer(200, {"rate_by_service": {"service:,env:": 1}})
        self.answer(202, {})


class Model(Quiet):
    def do_POST(self):
        self.body()
        if self.path.endswith("/chat/completions"):
            r = {"id": "chatcmpl-kiln1", "object": "chat.completion", "created": 1792800000, "model": "gpt-4o-mini-2024-07-18", "service_tier": "default",
                 "choices": [{"index": 0, "message": {"role": "assistant", "content": "Paris: 18C and clear."}, "finish_reason": "stop"}],
                 "usage": {"prompt_tokens": 1200, "completion_tokens": 40, "total_tokens": 1240, "prompt_tokens_details": {"cached_tokens": 1000}, "completion_tokens_details": {"reasoning_tokens": 10}}}
        elif self.path.endswith("/embeddings"):
            r = {"object": "list", "data": [{"object": "embedding", "index": 0, "embedding": [0.1, 0.2, 0.3]}], "model": "text-embedding-3-small", "usage": {"prompt_tokens": 8, "total_tokens": 8}}
        elif self.path.endswith("/messages"):
            r = {"id": "msg_kiln1", "type": "message", "role": "assistant", "model": "claude-opus-5-5", "content": [{"type": "text", "text": "Clear in Paris."}], "stop_reason": "end_turn",
                 "stop_sequence": None, "usage": {"input_tokens": 20, "output_tokens": 40, "cache_read_input_tokens": 1000, "cache_creation_input_tokens": 100,
                                                   "cache_creation": {"ephemeral_5m_input_tokens": 60, "ephemeral_1h_input_tokens": 40}}}
        else:
            return self.answer(404, {"error": "no such path"})
        self.answer(200, r)


for port, h in ((8126, Agent), (18080, Model)):
    s = socketserver.ThreadingTCPServer(("127.0.0.1", port), h)
    s.daemon_threads = True
    threading.Thread(target=s.serve_forever, daemon=True).start()

from ddtrace.llmobs import LLMObs
from ddtrace.llmobs.decorators import agent, tool, workflow

LLMObs.enable(ml_app="kiln-capture", integrations_enabled=True)
import anthropic
import openai

oc = openai.OpenAI(base_url="http://127.0.0.1:18080/v1", api_key="sk-kiln-capture")
ac = anthropic.Anthropic(base_url="http://127.0.0.1:18080", api_key="sk-ant-kiln-capture")


@tool(name="weather")
def weather(city):
    LLMObs.annotate(input_data={"city": city}, output_data="18C clear")
    return "18C clear"


@agent(name="planner")
def plan(q):
    r = oc.chat.completions.create(model="gpt-4o-mini", messages=[{"role": "system", "content": "You are terse."}, {"role": "user", "content": q}], temperature=0.2, max_tokens=100)
    weather("Paris")
    ac.messages.create(model="claude-opus-5-5", max_tokens=100, messages=[{"role": "user", "content": q}])
    oc.embeddings.create(model="text-embedding-3-small", input="weather in Paris")
    return r.choices[0].message.content


@workflow(name="answer", session_id="sess-kiln-1")
def answer(q):
    out = plan(q)
    LLMObs.annotate(input_data=q, output_data=out, tags={"team": "kiln"})
    return out, LLMObs.export_span()


out, exported = answer("Weather in Paris?")
LLMObs.submit_evaluation(label="accuracy", metric_type="score", value=0.8, span=exported, assessment="pass", reasoning="matches the forecast")
LLMObs.flush()
try:
    from ddtrace.trace import tracer
except ImportError:
    from ddtrace import tracer
tracer.flush()
time.sleep(3)
LLMObs.disable()
time.sleep(1)
print(MODE, "requests kept:", seq[0], file=sys.stderr)
